cmake_minimum_required(VERSION 3.16)
project(as2w_r214_deploy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(ONNXRUNTIME_ROOT "" CACHE PATH "Extracted ONNX Runtime package")
if(NOT ONNXRUNTIME_ROOT)
  message(FATAL_ERROR "Set ONNXRUNTIME_ROOT for R214")
endif()
find_package(yaml-cpp REQUIRED)
find_library(ort_library NAMES onnxruntime libonnxruntime.so.1
  PATHS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../../include/isaaclab/algorithms/algorithms.h")
  message(FATAL_ERROR "Copy this AS2-W target into the pinned unitree_cpp_deploy checkout")
endif()

add_library(as2w_r214 r214_controller.cpp)
target_include_directories(as2w_r214 PUBLIC . "${ONNXRUNTIME_ROOT}/include" ../../../include)
target_link_libraries(as2w_r214 PUBLIC "${ort_library}" yaml-cpp)

add_executable(as2w_r214_replay replay.cpp)
target_link_libraries(as2w_r214_replay PRIVATE as2w_r214)

option(AS2W_SDK2_SHADOW "Build the optional read-only SDK2 probe" OFF)
if(AS2W_SDK2_SHADOW)
  find_package(unitree_sdk2 CONFIG REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(as2w_r214_shadow shadow.cpp)
  target_link_libraries(as2w_r214_shadow PRIVATE as2w_r214 unitree_sdk2 Threads::Threads)
endif()

# R214 FSM publisher: event-driven state machine with remote control.
# Default OFF; never ship enabled. Read common/as2w_publish.cpp header gates first.
option(AS2W_R214_DIRECTION_TEST "Build the gated direction verification tool" OFF)
if(AS2W_R214_DIRECTION_TEST)
  find_package(unitree_sdk2 CONFIG REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(as2w_r214_direction_test r214_direction_test.cpp)
  target_link_libraries(as2w_r214_direction_test PRIVATE as2w_r214 unitree_sdk2 Threads::Threads)
endif()

option(AS2W_R214_PUBLISH "Build the gated R214 FSM LowCmd publisher" OFF)
option(AS2W_R214_STAND_TRIAL "Allow only the bounded A-to-fixed-PD stand trial" OFF)
option(AS2W_R214_ZERO_B_TRIAL "Allow only the bounded B zero-speed policy trial" OFF)
option(AS2W_R214_FULL_MOTION "Allow the supervised low-speed A/B/Y/X field session" OFF)
if(AS2W_R214_PUBLISH)
  find_package(unitree_sdk2 CONFIG REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(as2w_r214_publish ../common/as2w_publish.cpp)
  target_link_libraries(as2w_r214_publish PRIVATE as2w_r214 unitree_sdk2 Threads::Threads)
  if(AS2W_R214_STAND_TRIAL)
    target_compile_definitions(as2w_r214_publish PRIVATE AS2W_ENABLE_POLICY_STAND_TRIAL=1)
  endif()
  if(AS2W_R214_ZERO_B_TRIAL)
    target_compile_definitions(as2w_r214_publish PRIVATE AS2W_ENABLE_POLICY_ZERO_B_TRIAL=1)
  endif()
  if(AS2W_R214_FULL_MOTION)
    target_compile_definitions(as2w_r214_publish PRIVATE AS2W_ENABLE_POLICY_MOTION=1)
  endif()
  if(BUILD_TESTING)
    add_test(NAME r214-startup-frame-audit COMMAND as2w_r214_publish
      "${CMAKE_SOURCE_DIR}/models/r214/deploy.yaml"
      "${CMAKE_SOURCE_DIR}/config/r152_measured.map" --offline-audit)
  endif()
endif()
